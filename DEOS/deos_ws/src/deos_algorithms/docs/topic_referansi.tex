\subsection{JSON Formatları}

    "lateral_m": -0.6,    // negatif=sag
    "bbox_px": null        // LiDAR icin genellikle null
  }
]
\end{lstlisting}

\section{Algı Füzyon Topic'leri}
\label{sec:fusion}

\sectionhead{clrFusion}{Katman 3 — Perception Fusion}

\texttt{perception\_fusion\_node} 20\,Hz döngüsünde tüm algı modüllerini
(trafik ışığı, tabela, engel, park, slalom) \texttt{DecisionArbiter} üzerinden
birleştirir. Üretilen kararlar kontrol ve planlama katmanlarına iletilir.

\topiccard{clrFusion}
  {/deos/perception/fusion/emergency\_stop}
  {std\_msgs/Bool}
  {perception\_fusion\_node}
  {vehicle\_controller\_node}
  {\textbf{Acil dur kararı.} \texttt{true} ise araç derhal durmalıdır.
   Tetikleyiciler: kırmızı ışık ($\leq$3\,m), NO\_ENTRY tabelası,
   yaya ($\leq$5\,m), dinamik engel (dur fazı), tüm sensörler zaman aşımı.}

\topiccard{clrFusion}
  {/deos/perception/fusion/speed\_cap}
  {std\_msgs/Float32}
  {perception\_fusion\_node}
  {vehicle\_controller\_node}
  {\textbf{Hız üst limiti oranı.} Aralık: [0.0, 1.0].
   Tüm adaylar arasından minimum alınır (arbiter mantığı).
   0.0 = tam dur, 1.0 = kısıtlama yok.
   Formül: $v_{final} = \min(\text{speed\_cap}_{perc},\;\text{speed\_cap}_{fs},\;v_{plan},\;v_{lane}) \times v_{max}$.}

\topiccard{clrFusion}
  {/deos/perception/fusion/steering\_override}
  {std\_msgs/Float32}
  {perception\_fusion\_node}
  {vehicle\_controller\_node}
  {\textbf{Direksiyon geçersiz kılma.} Aralık: [-1.0, 1.0].
   Negatif = sola, pozitif = sağa. Yalnızca
   \texttt{has\_steering\_override=true} iken geçerlidir.
   Statik/dinamik engel geçişlerinde üretilir.}

\topiccard{clrFusion}
  {/deos/perception/fusion/has\_steering\_override}
  {std\_msgs/Bool}
  {perception\_fusion\_node}
  {vehicle\_controller\_node}
  {\textbf{Direksiyon geçersiz kılma aktif mi?}
   \texttt{true} iken \texttt{vehicle\_controller\_node} GPS planlama
   direksiyon referansı yerine \texttt{steering\_override} değerini kullanır.
   Şerit algısı ve engel kaçınma tazeyken aktifleşir.}

\topiccard{clrFusion}
  {/deos/perception/fusion/park\_complete}
  {std\_msgs/Bool}
  {perception\_fusion\_node}
  {mission\_planning\_node}
  {\textbf{Park manevra tamamlandı sinyali.}
   \texttt{ParkingLogic} PARKED fazına geçince \texttt{true} yayınlar.
   \texttt{mission\_planning\_node} bu sinyali alınca park waypoint'ini
   tamamlandı olarak işaretler ve sonraki waypoint'e geçer.}

\topiccard{clrFusion}
  {/deos/perception/fusion/turn\_permissions}
  {std\_msgs/String (JSON)}
  {perception\_fusion\_node}
  {mission\_planning\_node}
  {\textbf{Dönüş izinleri.} Trafik tabelalarından üretilir (NO\_LEFT\_TURN,
   MUST\_RIGHT, ROUNDABOUT, vb.).
   \texttt{mission\_planning\_node} yaklaşan waypoint'te bu kısıtlara göre
   Dijkstra ile yeniden rota hesaplar.
   JSON formatı için bkz. Bölüm~\ref{sec:fusion_json}.}

\topiccard{clrFusion}
  {/deos/perception/fusion/decision\_debug}
  {std\_msgs/String (JSON)}
  {perception\_fusion\_node}
  {mission\_planning\_node}
  {\textbf{Karar hata ayıklama bilgisi.} Tüm aday kararları, final kararı,
   \texttt{entry\_blocked} bayrağı (NO\_ENTRY tabelası) ve reason code listesi.
   \texttt{mission\_planning\_node} \texttt{road\_blocked} ve \texttt{entry\_blocked}
   sinyallerini bu JSON'dan okur.
   JSON formatı için bkz. Bölüm~\ref{sec:fusion_json}.}

\topiccard{clrFusion}
  {/deos/perception/fusion/green\_elapsed\_s}
  {std\_msgs/Float32}
  {perception\_fusion\_node}
  {vehicle\_controller\_node}
  {\textbf{Yeşil ışık geçen süre (saniye).}
   Yeşil ışık onaylandığı andan itibaren geçen süre.
   $-1.0$ = yeşil ışık yok / algılanmadı.
   \texttt{vehicle\_controller\_node} şu eşikleri izler:
   $\leq$5\,s = +40\,puan bandı, 5–30\,s = +20\,puan bandı,
   $>$30\,s = $-$20\,puan riski (log uyarısı üretir).}

\subsection{Fusion JSON Formatları}

      "name": "light",
      "emergency_stop": true,
      "speed_cap": 0.0,
      "reasons": ["light_must_stop"]
    }
  ],
  "final": {
    "emergency_stop": true,
    "speed_cap": 0.0,
    "has_steer_override": false,
    "steer_override": 0.0,
    "reasons": ["light_must_stop"]
  }
}
\end{lstlisting}

\sectionhead{clrPlanning}{Katman 4 — Mission Planning}

\texttt{mission\_planning\_node} GeoJSON görev dosyasını okur, GPS
callback'lerinde waypoint takibi yapar ve gerektiğinde Dijkstra ile
yeniden rota hesaplar (road\_blocked, entry\_blocked, dönüş kısıtı).

\topiccard{clrPlanning}
  {/deos/planning/steering\_ref}
  {std\_msgs/Float32}
  {mission\_planning\_node}
  {vehicle\_controller\_node, failsafe\_supervisor\_node}
  {\textbf{GPS bazlı direksiyon referansı.} Aralık: [-1.0, 1.0].
   Haversine azimuth + enine sapma (XTE) kombinasyonu.
   \texttt{vehicle\_controller\_node} şerit kontrolü taze değilse bu değeri kullanır.
   Failsafe doğrulama için izler.}

\topiccard{clrPlanning}
  {/deos/planning/speed\_limit}
  {std\_msgs/Float32}
  {mission\_planning\_node}

\subsection{Acil Dur Zinciri}

\begin{center}
\begin{tabular}{@{}p{4.5cm}p{6cm}p{4cm}@{}}
\toprule
\textbf{Kaynak} & \textbf{Topic} & \textbf{Alıcı} \\
\midrule
\texttt{perception\_fusion\_node}  & \texttt{.../fusion/emergency\_stop}        & \texttt{vehicle\_controller} \\
\texttt{failsafe\_supervisor\_node}& \texttt{.../failsafe/out/emergency\_stop}  & \texttt{vehicle\_controller} \\

\bottomrule
\end{tabular}
\end{center}

İki kaynak OR mantığıyla birleşir: herhangi biri \texttt{true} ise araç durur.

\subsection{Hız Kısıtı Zinciri}

\[
v_{final} = \min\!\bigl(
  v_{plan},\;
  v_{lane},\;
  \text{cap}_{perception},\;
  \text{cap}_{failsafe}
\bigr) \times v_{max}
\]

Şerit ihlali aktifken failsafe cap en fazla 0.5 olur:
$\text{cap}_{failsafe} = \min\bigl(\text{cap}_{fsm},\; 0.5\bigr)$ (ihlal varsa).

\subsection{Direksiyon Öncelik Sırası}

\begin{center}
\begin{tabular}{@{}p{0.5cm}p{4.5cm}p{5cm}p{4cm}@{}}
\toprule

\midrule

3 & diğer                              & \texttt{planning/steering\_ref}     & \texttt{mission\_planning} \\

\subsection{Replanning Tetikleyicileri}

\texttt{mission\_planning\_node} Dijkstra ile yeniden rota hesapladığı durumlar:

\begin{itemize}[noitemsep, topsep=4pt]
  \item \texttt{decision\_debug.final.reasons} içinde \texttt{"road\_blocked"} var
  \item \texttt{decision\_debug.entry\_blocked = true} (NO\_ENTRY tabelası)
  \item Sonraki waypoint'e $\leq$8\,m kaldıyken aktif dönüş kısıtı var
        (\texttt{turn\_permissions})
\end{itemize}

Replanning sonrası \texttt{road\_blocked} ve \texttt{entry\_blocked} bayrakları
sıfırlanır (yeni rotada aynı kısıt tekrar algılanana kadar beklenmez).

\subsection{Başlangıç Sırası (Boot Sequence)}

\begin{enumerate}[noitemsep, topsep=4pt]
  \item Systemd \texttt{deos-bringup.service} → Docker Compose başlatır
  \item Container içinde \texttt{colcon build} (install/ yoksa)
  \item \texttt{ros2 launch vehicle\_bringup main.launch.py}
  \item \texttt{ros2 bag record} otomatik başlar (\texttt{/ros2\_ws/bags/YYYY-MM-DD\_HH-MM-SS/})
  \item \texttt{deos-health.service} 10\,s aralıkla 11 kritik node'u izler
  \item 3 ardışık hata → \texttt{deos-bringup.service} yeniden başlatılır
\end{enumerate}
